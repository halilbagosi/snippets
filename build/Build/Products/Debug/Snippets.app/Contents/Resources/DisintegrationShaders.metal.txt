#include <metal_stdlib>
using namespace metal;

// ---------- hashing helpers ----------

float hash12(float2 p) {
    float3 p3 = fract(float3(p.xyx) * 0.1031);
    p3 += dot(p3, p3.yzx + 33.33);
    return fract((p3.x + p3.y) * p3.z);
}

float2 hash22(float2 p) {
    float3 p3 = fract(float3(p.xyx) * float3(0.1031, 0.11369, 0.13787));
    p3 += dot(p3, p3.yzx + 19.19);
    return fract(float2((p3.x + p3.y) * p3.z, (p3.x + p3.z) * p3.y));
}

// Simple value noise for organic dissolve boundary
float valueNoise(float2 p) {
    float2 i = floor(p);
    float2 f = fract(p);
    f = f * f * (3.0 - 2.0 * f); // smoothstep interpolation

    float a = hash12(i);
    float b = hash12(i + float2(1.0, 0.0));
    float c = hash12(i + float2(0.0, 1.0));
    float d = hash12(i + float2(1.0, 1.0));

    return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
}

// Multi-octave noise for richer dissolve edges
float fbm(float2 p) {
    float value = 0.0;
    float amplitude = 0.5;
    for (int i = 0; i < 3; i++) {
        value += amplitude * valueNoise(p);
        p *= 2.17;
        amplitude *= 0.5;
    }
    return value;
}

// ---------- vertex ----------

struct VertexOut {
    float4 position [[position]];
    float2 uv;
};

vertex VertexOut vertex_main(uint vertexID [[vertex_id]]) {
    float2 positions[3] = {
        float2(-1.0, -1.0),
        float2(3.0, -1.0),
        float2(-1.0, 3.0)
    };
    float2 position = positions[vertexID];

    VertexOut out;
    out.position = float4(position, 0.0, 1.0);
    out.uv = position * 0.5 + 0.5;
    return out;
}

// ---------- uniforms ----------

struct Uniforms {
    float4 sizeAndProgress; // x = width, y = height, z = progress
    float4 accentColor;
};

// ---------- fragment ----------

fragment half4 fragment_main(
    VertexOut in [[stage_in]],
    texture2d<float> inputTexture [[texture(0)]],
    sampler inputSampler [[sampler(0)]],
    constant Uniforms &uniforms [[buffer(0)]]
) {
    float2 size = max(uniforms.sizeAndProgress.xy, float2(1.0));
    float progress = clamp(uniforms.sizeAndProgress.z, 0.0, 1.0);
    float2 position = in.uv * size;
    float2 normalizedPosition = position / size;

    // --- Particle grid ---
    // Smaller cells = finer particles for a dusty dissolve look
    float cellSize = mix(2.5, 4.0, smoothstep(0.1, 0.7, progress));
    float2 cell = floor(position / cellSize);
    float seed = hash12(cell);
    float2 random = hash22(cell + 17.0);

    // --- Dissolve ordering ---
    // Use fbm noise so the dissolve boundary is organic, not blocky
    float noiseVal = fbm(cell * 0.25 + float2(3.7, 1.3));
    float directional = normalizedPosition.x * 0.20 + (1.0 - normalizedPosition.y) * 0.10;
    float dissolveOrder = clamp(noiseVal * 0.55 + seed * 0.15 + directional, 0.0, 1.0);

    // Wider transition band for smoother dissolve edge
    float release = smoothstep(dissolveOrder - 0.22, dissolveOrder + 0.22, progress);
    float erase  = smoothstep(dissolveOrder + 0.05, dissolveOrder + 0.28, progress);

    // --- Particle drift with ease-out physics ---
    float easedRelease = 1.0 - (1.0 - release) * (1.0 - release); // quadratic ease-out
    float2 direction = normalize(float2(0.4 + random.x * 0.8, -1.0 + (random.y - 0.5) * 0.6));

    // Gentler flutter with dampening
    float flutter = sin(progress * 18.0 + seed * 20.0) * 3.5 * easedRelease * (1.0 - progress * 0.5);

    // Drift distance ramps up with ease-out, less extreme range
    float driftDistance = easedRelease * (12.0 + seed * 70.0) * (0.3 + progress * 0.7);

    float2 drift = direction * driftDistance + float2(flutter, -abs(flutter) * 0.25);

    // --- Per-particle scale (shrink as they fly off) ---
    float particleScale = mix(1.0, 0.3, easedRelease * easedRelease);
    float2 cellCenter = (cell + 0.5) * cellSize;
    float2 localOffset = position - cellCenter;
    float2 scaledPosition = cellCenter + localOffset / particleScale;

    float2 samplePosition = scaledPosition - drift;
    float2 sampleUV = samplePosition / size;

    // Out-of-bounds check
    if (sampleUV.x < 0.0 || sampleUV.x > 1.0 || sampleUV.y < 0.0 || sampleUV.y > 1.0) {
        return half4(0.0);
    }

    float4 sampled = inputTexture.sample(inputSampler, sampleUV);
    half4 color = half4(half(sampled.x), half(sampled.y), half(sampled.z), half(sampled.w));
    half3 accent = half3(half(uniforms.accentColor.x), half(uniforms.accentColor.y), half(uniforms.accentColor.z));

    // --- Edge glow at dissolve frontier ---
    // Softer, wider glow that fades out gracefully
    float edgeBand = abs(progress - dissolveOrder);
    float edgeGlow = smoothstep(0.18, 0.0, edgeBand) *
                     smoothstep(0.0, 0.08, progress) *      // fade in at start
                     (1.0 - smoothstep(0.80, 1.0, progress)); // fade out at end

    // Subtle ember darkening on late-stage particles
    float ember = smoothstep(0.60, 0.95, progress) *
                  hash12(floor(samplePosition / 3.0) + float2(seed)) *
                  0.18;

    // --- Compose final color ---
    color.rgb = mix(color.rgb, accent, half(edgeGlow * 0.40));
    color.rgb += half3(half(edgeGlow * 0.18));
    color.rgb = mix(color.rgb, half3(0.03h, 0.03h, 0.035h), half(ember));

    // Smooth alpha: combine dissolve erase with per-particle fade and global tail fade
    float particleFade = 1.0 - easedRelease * easedRelease * 0.5; // particles become translucent
    float tailFade = 1.0 - smoothstep(0.88, 1.0, progress);       // everything fades at end
    color.a *= half((1.0 - erase) * particleFade * tailFade);

    return color;
}
