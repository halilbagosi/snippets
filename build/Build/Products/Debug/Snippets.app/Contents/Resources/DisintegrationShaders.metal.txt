#include <metal_stdlib>
using namespace metal;

// ---------- hashing helpers ----------

float hash12(float2 p) {
    float3 p3 = fract(float3(p.xyx) * 0.1031);
    p3 += dot(p3, p3.yzx + 33.33);
    return fract((p3.x + p3.y) * p3.z);
}

float2 hash22(float2 p) {
    float3 p3 = fract(float3(p.xyx) * float3(0.1031, 0.11369, 0.13787));
    p3 += dot(p3, p3.yzx + 19.19);
    return fract(float2((p3.x + p3.y) * p3.z, (p3.x + p3.z) * p3.y));
}

// Simple value noise for organic dissolve boundary
float valueNoise(float2 p) {
    float2 i = floor(p);
    float2 f = fract(p);
    f = f * f * (3.0 - 2.0 * f); // smoothstep interpolation

    float a = hash12(i);
    float b = hash12(i + float2(1.0, 0.0));
    float c = hash12(i + float2(0.0, 1.0));
    float d = hash12(i + float2(1.0, 1.0));

    return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
}

// Multi-octave noise for richer dissolve edges
float fbm(float2 p) {
    float value = 0.0;
    float amplitude = 0.5;
    for (int i = 0; i < 3; i++) {
        value += amplitude * valueNoise(p);
        p *= 2.17;
        amplitude *= 0.5;
    }
    return value;
}

// ---------- vertex ----------

struct VertexOut {
    float4 position [[position]];
    float2 uv;
};

vertex VertexOut vertex_main(uint vertexID [[vertex_id]]) {
    float2 positions[3] = {
        float2(-1.0, -1.0),
        float2(3.0, -1.0),
        float2(-1.0, 3.0)
    };
    float2 position = positions[vertexID];

    VertexOut out;
    out.position = float4(position, 0.0, 1.0);
    out.uv = position * 0.5 + 0.5;
    return out;
}

// ---------- uniforms ----------

struct Uniforms {
    float4 sizeAndProgress; // x = width, y = height, z = progress
    float4 accentColor;
};

// ---------- fragment ----------
//
// Design notes (smoothness):
// * The particle grid uses a CONSTANT cell size. A progress-dependent cell
//   size re-bins every pixel each frame, which visibly reshuffles particle
//   identities mid-flight — a major source of the old choppiness.
// * All per-particle randomness (seed, direction, timing) is keyed to the
//   stable cell id, never to coordinates that move over time, so nothing
//   flickers frame-over-frame.
// * Every particle animates on its own local time `t` (a smooth 0→1 window
//   of the global progress). The dissolve frontier sweeps across the card
//   while each released particle moves through one continuous, eased arc.

fragment half4 fragment_main(
    VertexOut in [[stage_in]],
    texture2d<float> inputTexture [[texture(0)]],
    sampler inputSampler [[sampler(0)]],
    constant Uniforms &uniforms [[buffer(0)]]
) {
    float2 size = max(uniforms.sizeAndProgress.xy, float2(1.0));
    float progress = clamp(uniforms.sizeAndProgress.z, 0.0, 1.0);
    float2 position = in.uv * size;
    float2 normalizedPosition = position / size;

    // --- Stable particle grid ---
    const float cellSize = 3.0;
    float2 cell = floor(position / cellSize);
    float seed = hash12(cell);
    float2 random = hash22(cell + 17.0);

    // --- Dissolve ordering ---
    // Organic fbm wave with a light per-particle jitter, gently biased so the
    // frontier sweeps across the card instead of appearing everywhere at once.
    float noiseVal = fbm(cell * 0.11 + float2(3.7, 1.3));
    float directional = normalizedPosition.x * 0.16 + (1.0 - normalizedPosition.y) * 0.10;
    float order = clamp(noiseVal * 0.60 + seed * 0.10 + directional, 0.0, 0.97);

    // --- Per-particle local time ---
    // Each particle lives through its own smooth 0→1 life once the frontier
    // reaches it. A longer life window keeps individual motion unhurried.
    const float life = 0.38;
    float t = smoothstep(order, order + life, progress);

    if (t >= 1.0) {
        return half4(0.0);
    }

    // --- Motion: eased drift up and outward, low-frequency sway ---
    float easeOut = 1.0 - (1.0 - t) * (1.0 - t);
    float2 direction = normalize(float2(0.35 + random.x * 0.75, -1.0 + (random.y - 0.5) * 0.5));
    float travel = (16.0 + seed * 60.0) * easeOut;
    // At most ~one gentle sway cycle per particle life (the old 18 Hz flutter
    // read as jitter); phase and rate are seed-randomized so cells desync.
    float sway = sin(t * (2.4 + seed * 2.2) + seed * 6.2831) * 5.0 * t;
    float2 drift = direction * travel + float2(sway, sway * -0.35);

    // --- Gentle shrink as particles fly (strong shrink magnifies
    //     cross-cell sampling shimmer, so keep it modest) ---
    float particleScale = mix(1.0, 0.55, t * t);
    float2 cellCenter = (cell + 0.5) * cellSize;
    float2 localOffset = position - cellCenter;
    float2 samplePosition = cellCenter + localOffset / particleScale - drift;
    float2 sampleUV = samplePosition / size;

    if (sampleUV.x < 0.0 || sampleUV.x > 1.0 || sampleUV.y < 0.0 || sampleUV.y > 1.0) {
        return half4(0.0);
    }

    float4 sampled = inputTexture.sample(inputSampler, sampleUV);
    half4 color = half4(half(sampled.x), half(sampled.y), half(sampled.z), half(sampled.w));
    half3 accent = half3(half(uniforms.accentColor.x), half(uniforms.accentColor.y), half(uniforms.accentColor.z));

    // --- Frontier glow ---
    // A soft accent shimmer riding just behind the dissolve wave; band position
    // depends only on (progress - order), which evolves smoothly.
    float edgeBand = progress - order;
    float frontierGlow = smoothstep(0.14, 0.0, abs(edgeBand - 0.03)) *
                         smoothstep(0.0, 0.06, progress) *
                         (1.0 - smoothstep(0.82, 1.0, progress));

    // --- Ember dimming, keyed to the STABLE cell seed (the old version
    //     hashed a moving coordinate, producing per-frame sparkle noise) ---
    float ember = smoothstep(0.35, 0.85, t) * seed * 0.16;

    color.rgb = mix(color.rgb, accent, half(frontierGlow * 0.42 * float(color.a)));
    color.rgb += half3(half(frontierGlow * 0.16 * float(color.a)));
    color.rgb = mix(color.rgb, half3(0.04h, 0.04h, 0.045h), half(ember));

    // --- Fade ---
    // Particles stay solid through the front half of their life, then fade
    // smoothly out; a thin global tail catches any stragglers. The output is
    // premultiplied (blending uses .one / .oneMinusSourceAlpha), so the fade
    // scales all four channels — fading alpha alone leaves a hot additive
    // ghost, part of the old "unnatural" look.
    float fade = 1.0 - smoothstep(0.45, 1.0, t);
    float tail = 1.0 - smoothstep(0.94, 1.0, progress);
    color *= half(fade * tail);

    return color;
}
